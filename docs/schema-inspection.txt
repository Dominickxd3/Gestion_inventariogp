sp_sincronizar_usuarios_erp 

/* ============================================================
   1. SINCRONIZAR USUARIOS ERP
   ============================================================ */

CREATE PROCEDURE dbo.sp_sincronizar_usuarios_erp
AS
BEGIN

    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    -----------------------------------------------------------
    -- ACTUALIZAR USUARIOS EXISTENTES
    -----------------------------------------------------------

    UPDATE LOCAL
    SET
        LOCAL.user_fullname = ERP.User_Fullname,
        LOCAL.user_email = ERP.User_Email,
        LOCAL.user_estado_erp = ERP.User_Estado,
        LOCAL.ultima_sincronizacion = SYSDATETIME(),
        LOCAL.fecha_actualizacion = SYSDATETIME()

    FROM dbo.usuarios_sistema LOCAL

    INNER JOIN
        [DB_GP_Trabajos_TEST].[dbo].[vw_ERP_Usuarios] ERP
        ON ERP.User_Logon COLLATE DATABASE_DEFAULT =
           LOCAL.user_logon COLLATE DATABASE_DEFAULT;


    -----------------------------------------------------------
    -- INSERTAR USUARIOS NUEVOS
    -----------------------------------------------------------

    INSERT INTO dbo.usuarios_sistema
    (
        user_logon,
        user_fullname,
        user_email,
        user_estado_erp,
        permitir_acceso,
        estado_local,
        ultima_sincronizacion
    )

    SELECT DISTINCT

        ERP.User_Logon,
        ERP.User_Fullname,
        ERP.User_Email,
        ERP.User_Estado,

        0,
        1,

        SYSDATETIME()

    FROM
        [DB_GP_Trabajos_TEST].[dbo].[vw_ERP_Usuarios] ERP

    WHERE
        ERP.User_Logon IS NOT NULL

        AND NOT EXISTS
        (
            SELECT 1
            FROM dbo.usuarios_sistema LOCAL

            WHERE LOCAL.user_logon COLLATE DATABASE_DEFAULT =
                  ERP.User_Logon COLLATE DATABASE_DEFAULT
        );


    -----------------------------------------------------------
    -- RESULTADO
    -----------------------------------------------------------

    SELECT
        COUNT(*) AS total_usuarios_locales,

        SUM
        (
            CASE
                WHEN permitir_acceso = 1 THEN 1
                ELSE 0
            END
        ) AS usuarios_autorizados,

        SYSDATETIME() AS fecha_sincronizacion

    FROM dbo.usuarios_sistema;

END;

sp_asignar_activo 

/* ============================================================
   3. ASIGNAR ACTIVO
   ============================================================ */

CREATE PROCEDURE dbo.sp_asignar_activo

    @id_activo INT,
    @id_trabajador INT,
    @estado_entrega VARCHAR(150) = NULL,
    @motivo VARCHAR(300) = NULL,
    @observacion VARCHAR(1500) = NULL,
    @id_usuario INT = NULL

AS
BEGIN

    SET NOCOUNT ON;
    SET XACT_ABORT ON;


    IF EXISTS
    (
        SELECT 1
        FROM dbo.asignaciones_activo
        WHERE
            id_activo = @id_activo
            AND activo = 1
    )
    BEGIN

        RAISERROR(
            'El activo ya tiene una asignación activa.',
            16,
            1
        );

        RETURN;

    END;


    BEGIN TRY

        BEGIN TRANSACTION;


        INSERT INTO dbo.asignaciones_activo
        (
            id_activo,
            id_trabajador,
            estado_entrega,
            motivo_entrega,
            observacion_entrega,
            usuario_entrega
        )
        VALUES
        (
            @id_activo,
            @id_trabajador,
            @estado_entrega,
            @motivo,
            @observacion,
            @id_usuario
        );


        UPDATE dbo.activos

        SET
            id_estado_activo =
            (
                SELECT id_estado_activo
                FROM dbo.estados_activo
                WHERE codigo = 'ASIGNADO'
            ),

            fecha_actualizacion = SYSDATETIME()

        WHERE id_activo = @id_activo;


        INSERT INTO dbo.movimientos_activo
        (
            id_activo,
            tipo_movimiento,
            id_trabajador_destino,
            descripcion,
            id_usuario
        )
        VALUES
        (
            @id_activo,
            'ASIGNACION',
            @id_trabajador,
            'Activo asignado al trabajador',
            @id_usuario
        );


        COMMIT TRANSACTION;

    END TRY

    BEGIN CATCH

        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;

        DECLARE @MensajeError NVARCHAR(4000);

        SET @MensajeError = ERROR_MESSAGE();

        RAISERROR(@MensajeError,16,1);

    END CATCH;

END;

sp_devolver_activo 

/* ============================================================
   4. DEVOLVER ACTIVO
   ============================================================ */

CREATE PROCEDURE dbo.sp_devolver_activo

    @id_activo INT,
    @estado_devolucion VARCHAR(150) = NULL,
    @motivo VARCHAR(300) = NULL,
    @observacion VARCHAR(1500) = NULL,
    @id_usuario INT = NULL

AS
BEGIN

    SET NOCOUNT ON;
    SET XACT_ABORT ON;


    DECLARE @id_trabajador INT;


    SELECT
        @id_trabajador = id_trabajador

    FROM dbo.asignaciones_activo

    WHERE
        id_activo = @id_activo
        AND activo = 1;


    IF @id_trabajador IS NULL
    BEGIN

        RAISERROR(
            'El activo no posee una asignación activa.',
            16,
            1
        );

        RETURN;

    END;


    BEGIN TRY

        BEGIN TRANSACTION;


        UPDATE dbo.asignaciones_activo

        SET
            fecha_devolucion = SYSDATETIME(),

            estado_devolucion = @estado_devolucion,

            motivo_devolucion = @motivo,

            observacion_devolucion = @observacion,

            usuario_devolucion = @id_usuario,

            activo = 0

        WHERE
            id_activo = @id_activo
            AND activo = 1;


        UPDATE dbo.activos

        SET
            id_estado_activo =
            (
                SELECT id_estado_activo
                FROM dbo.estados_activo
                WHERE codigo = 'DISPONIBLE'
            ),

            fecha_actualizacion = SYSDATETIME()

        WHERE id_activo = @id_activo;


        INSERT INTO dbo.movimientos_activo
        (
            id_activo,
            tipo_movimiento,
            id_trabajador_origen,
            descripcion,
            id_usuario
        )
        VALUES
        (
            @id_activo,
            'DEVOLUCION',
            @id_trabajador,
            'Activo devuelto al área de Sistemas',
            @id_usuario
        );


        COMMIT TRANSACTION;

    END TRY

    BEGIN CATCH

        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;

        DECLARE @MensajeError NVARCHAR(4000);

        SET @MensajeError = ERROR_MESSAGE();

        RAISERROR(@MensajeError,16,1);

    END CATCH;

END;

sp_validar_acceso_sistema 
CREATE PROCEDURE dbo.sp_validar_acceso_sistema
    @Usuario VARCHAR(100)
AS
BEGIN
    SET NOCOUNT ON;

    IF NOT EXISTS
    (
        SELECT 1
        FROM dbo.usuarios_sistema
        WHERE user_logon = @Usuario
    )
    BEGIN
        SELECT
            0 AS Success,
            'Usuario no registrado en Gestión TI' AS Message;
        RETURN;
    END;


    -- 0 = ACTIVO
    -- 1 = INACTIVO
    IF EXISTS
    (
        SELECT 1
        FROM dbo.usuarios_sistema
        WHERE
            user_logon = @Usuario
            AND ISNULL(user_estado_erp,1) <> 0
    )
    BEGIN
        SELECT
            0 AS Success,
            'Usuario inactivo en el sistema corporativo' AS Message;
        RETURN;
    END;


    IF EXISTS
    (
        SELECT 1
        FROM dbo.usuarios_sistema
        WHERE
            user_logon = @Usuario
            AND estado_local = 0
    )
    BEGIN
        SELECT
            0 AS Success,
            'Usuario deshabilitado en Gestión TI' AS Message;
        RETURN;
    END;


    IF EXISTS
    (
        SELECT 1
        FROM dbo.usuarios_sistema
        WHERE
            user_logon = @Usuario
            AND permitir_acceso = 0
    )
    BEGIN
        SELECT
            0 AS Success,
            'El usuario no tiene acceso autorizado a Gestión TI' AS Message;
        RETURN;
    END;


    IF EXISTS
    (
        SELECT 1
        FROM dbo.usuarios_sistema
        WHERE
            user_logon = @Usuario
            AND id_rol IS NULL
    )
    BEGIN
        SELECT
            0 AS Success,
            'El usuario no tiene un rol asignado' AS Message;
        RETURN;
    END;


    UPDATE dbo.usuarios_sistema
    SET ultimo_acceso = SYSDATETIME()
    WHERE user_logon = @Usuario;


    SELECT
        1 AS Success,
        'Acceso autorizado' AS Message,

        U.id_usuario_sistema,
        U.user_logon,
        U.user_fullname,
        U.user_email,

        R.id_rol,
        R.codigo AS rol_codigo,
        R.nombre AS rol

    FROM dbo.usuarios_sistema U

    INNER JOIN dbo.roles R
        ON R.id_rol = U.id_rol

    WHERE U.user_logon = @Usuario;
END;

-- SP: Login de usuario
CREATE PROCEDURE sp_login_usuario
    @Usuario VARCHAR(50),
    @Contrasena VARCHAR(255)
AS
BEGIN
    IF NOT EXISTS (SELECT 1 FROM TBL_USUARIOS WHERE Usuario = @Usuario)
    BEGIN
        SELECT 0 AS Success, 'Usuario no encontrado' AS Message
        RETURN
    END
    
    IF EXISTS (SELECT 1 FROM TBL_USUARIOS WHERE Usuario = @Usuario AND Bloqueado = '1')
    BEGIN
        SELECT 0 AS Success, 'Usuario bloqueado. Contacte al administrador' AS Message
        RETURN
    END
    
    IF EXISTS (SELECT 1 FROM TBL_USUARIOS WHERE Usuario = @Usuario AND Activo = '2')
    BEGIN
        SELECT 0 AS Success, 'Usuario inactivo' AS Message
        RETURN
    END
    
    IF EXISTS (SELECT 1 FROM TBL_USUARIOS WHERE Usuario = @Usuario AND Contrasena = @Contrasena)
    BEGIN
        UPDATE TBL_USUARIOS 
        SET UltimoAcceso = GETDATE(), IntentosFallidos = 0
        WHERE Usuario = @Usuario
        
        SELECT 
            1 AS Success,
            'Login exitoso' AS Message,
            IdUsuario, Usuario, NombreCompleto, Email, Rol, UltimoAcceso
        FROM TBL_USUARIOS
        WHERE Usuario = @Usuario
    END
    ELSE
    BEGIN
        UPDATE TBL_USUARIOS 
        SET IntentosFallidos = IntentosFallidos + 1,
            Bloqueado = CASE WHEN IntentosFallidos >= 4 THEN '1' ELSE '2' END
        WHERE Usuario = @Usuario
        
        SELECT 0 AS Success, 'Contraseña incorrecta' AS Message
    END
END
